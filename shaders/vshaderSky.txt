#version 330 core
layout (location = 0) in vec3 aPos;

out vec3 vDir;

uniform mat4 projection;		
uniform mat4 view;

void main()
{
    // Remove translation from view so the cube stays centered on the camera
    mat4 rotView = mat4(mat3(view));
    vDir = aPos;
    gl_Position = projection * rotView * vec4(aPos, 1.0);
}
